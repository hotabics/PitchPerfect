#### **WPM Calculation Algorithm**

```python
def calculate_wpm(transcript, start_time, current_time):
    """
    Calculate words per minute in real-time
    Uses rolling 30-second window for accuracy
    """
    words = transcript.split()
    time_elapsed = (current_time - start_time).total_seconds() / 60
    
    if time_elapsed < 0.5:  # Less than 30 seconds
        return 0  # Not enough data
    
    wpm = len(words) / time_elapsed
    
    # Rolling average (last 30 seconds)
    recent_words = [w for w in words if w.timestamp > (current_time - 30)]
    rolling_wpm = len(recent_words) * 2  # 30 sec * 2 = 60 sec
    
    return rolling_wpm
```

#### **Traffic Light Logic**

```javascript
function determineState(metrics) {
  const { wpm, pauseDuration, fillerCount, monologueDuration } = metrics;
  
  // RED: Critical intervention needed
  if (wpm > 180 || monologueDuration > 90 || pauseDuration > 8) {
    return {
      color: 'RED',
      icon: wpm > 180 ? '↓' : monologueDuration > 90 ? '?' : '→',
      haptic: 'double_pulse'
    };
  }
  
  // YELLOW: Caution
  if (wpm > 160 || pauseDuration > 3 || fillerCount > 3) {
    return {
      color: 'YELLOW',
      icon: null,
      haptic: 'single_pulse'
    };
  }
  
  // GREEN: Optimal
  return {
    color: 'GREEN',
    icon: null,
    haptic: null
  };
}
```
